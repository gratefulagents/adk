package anthropic

import (
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"net/http/httptest"
	"os"
	"path/filepath"
	"strings"
	"testing"
	"time"
)

func TestADKAnthropicUsageGolden(t *testing.T) {
	updates := [][]string{
		{},
		{`{"output_tokens":42}`},
		{`{"input_tokens":0,"output_tokens":0,"cache_read_input_tokens":0,"cache_creation_input_tokens":0}`},
		{`{"input_tokens":12,"output_tokens":7,"cache_read_input_tokens":8,"cache_creation_input_tokens":3}`},
		{`{"output_tokens":0}`},
		{`{"input_tokens":null,"output_tokens":null,"cache_read_input_tokens":null,"cache_creation_input_tokens":null}`},
		{`{}`},
		{`null`},
		{``},
		{`{"input_tokens":12,"output_tokens":7,"cache_read_input_tokens":8,"cache_creation_input_tokens":3}`, `{"input_tokens":0,"output_tokens":0,"cache_read_input_tokens":0,"cache_creation_input_tokens":0}`},
		{`{"output_tokens":42}`, `{"output_tokens":0}`, `{"output_tokens":51}`},
	}
	var rows []map[string]any
	for _, sequence := range updates {
		events := []string{
			`{"type":"message_start","message":{"id":"m","type":"message","role":"assistant","model":"fixture","content":[],"usage":{"input_tokens":1200,"output_tokens":5,"cache_read_input_tokens":800,"cache_creation_input_tokens":50}}}`,
			`{"type":"content_block_start","index":0,"content_block":{"type":"text","text":""}}`,
			`{"type":"content_block_delta","index":0,"delta":{"type":"text_delta","text":"ok"}}`,
			`{"type":"content_block_stop","index":0}`,
		}
		for _, usage := range sequence {
			event := `{"type":"message_delta","delta":{"stop_reason":"end_turn"}`
			if usage != "" {
				event += `,"usage":` + usage
			}
			events = append(events, event+`}`)
		}
		events = append(events, `{"type":"message_stop"}`)
		var sse strings.Builder
		for _, event := range events {
			var envelope struct {
				Type string `json:"type"`
			}
			if err := json.Unmarshal([]byte(event), &envelope); err != nil {
				t.Fatal(err)
			}
			fmt.Fprintf(&sse, "event: %s\ndata: %s\n\n", envelope.Type, event)
		}
		server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
			w.Header().Set("Content-Type", "text/event-stream")
			fmt.Fprint(w, sse.String())
		}))
		ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		client := NewClient("fixture", WithBaseURL(server.URL))
		request := CreateMessageRequest{Model: "fixture", MaxTokens: 100, Messages: []Message{{Role: RoleUser, Content: []ContentBlock{NewTextBlock("hi")}}}}
		complete, err := client.CreateMessage(ctx, request)
		if err != nil {
			t.Fatal(err)
		}
		stream, err := client.CreateMessageStream(ctx, request)
		if err != nil {
			t.Fatal(err)
		}
		collected, err := stream.CollectResponse()
		if err != nil {
			t.Fatal(err)
		}
		stream.Close()
		cancel()
		server.Close()
		rows = append(rows, map[string]any{"events": events, "complete_usage": complete.Usage, "stream_usage": collected.Usage})
	}
	encoded, err := json.MarshalIndent(rows, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(os.Getenv("ADK_PROVIDER_FIXTURE_DIR"), "anthropic-usage.json"), append(encoded, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
}
