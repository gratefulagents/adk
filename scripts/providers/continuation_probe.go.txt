func TestADKContinuationGolden(t *testing.T) {
    items, err := toResponseInputItems([]anthropic.Message{{
        Role: anthropic.RoleAssistant,
        Content: []anthropic.ContentBlock{
            {Type: "thinking", ID: "r", Thinking: "summary", EncryptedContent: "reasoning-opaque"},
            anthropic.NewCompactionBlock("c", "compact-opaque", "openai"),
        },
    }})
    if err != nil { t.Fatal(err) }
    request := anthropic.CreateMessageRequest{Model: "gpt-5.6", MaxTokens: 16384,
        System: []anthropic.SystemBlock{{Type: "text", Text: "Be precise"}},
        Messages: []anthropic.Message{{Role: anthropic.RoleUser, Content: []anthropic.ContentBlock{{Type: "text", Text: "hello"}}}},
        ReasoningEffort: "high"}
    responsesRequest, err := toResponseParams(request)
    if err != nil { t.Fatal(err) }
    chatRequest, err := toChatRequest(request)
    if err != nil { t.Fatal(err) }
    details := json.RawMessage(`[{"type":"reasoning.encrypted","data":"opaque","index":0}]`)
    golden := map[string]any{
        "responses_request": responsesRequest,
        "chat_request": chatRequest,
        "source_revision": "1dc92b73900fac74dc357a938e4b5eee6392b418",
        "responses_input": items,
        "reasoning_details_json": details,
        "reasoning_details_signature": encodeOpenRouterReasoningDetails(details),
    }
    var outputSchemas []map[string]any
    for _, raw := range []string{`null`, `{}`, `"schema"`, `7`, `[]`, `true`, `false`, `{"type":27}`, `{"x":-0}`, `{"x":9007199254740993}`, `{"x":1e400}`, `{"x":1e400,"x":1}`, `{"x":"\ud800"}`, `{"x":1,"x":2}`} {
        schema := &anthropic.OutputSchema{Name:"result", Schema:json.RawMessage(raw), Strict:true}
        normalized, err := outputSchemaMap(schema)
        _, chatErr := chatResponseFormat(schema)
        _, responsesErr := responseFormatFromSchema(schema)
        if (err != nil) != (chatErr != nil) || (err != nil) != (responsesErr != nil) { t.Fatal("schema acceptance differs between helpers") }
        row := map[string]any{"raw":raw, "error":err != nil}
        if err == nil {
            encoded, marshalErr := json.Marshal(normalized)
            if marshalErr != nil { t.Fatal(marshalErr) }
            row["schema_json"] = string(encoded)
        }
        outputSchemas = append(outputSchemas, row)
    }
    golden["output_schemas"] = outputSchemas
    var normalizedArguments []map[string]string
    for _, raw := range []string{
        "", " \t\n", "\u0085{}\u00a0", ` {"q":1} `, `{"q":"\ud800"}`, `{"q":"\udc00"}`,
        `{"q":"\ud83d\ude00"}`, `{"q":1e999}`, `{"q":1,"q":2}`, `null`, `42`,
        `{"q":}`, `{} {}`, strings.Repeat("[", 128) + "0" + strings.Repeat("]", 128),
        strings.Repeat("[", 10000) + "0" + strings.Repeat("]", 10000),
        strings.Repeat("[", 10001) + "0" + strings.Repeat("]", 10001),
    } {
        normalizedArguments = append(normalizedArguments, map[string]string{
            "input": raw, "normalized": string(normalizeArguments(raw)),
        })
    }
    golden["normalized_arguments"] = normalizedArguments
    var bufferedDetails []map[string]any
    var streamedDetails []map[string]any
    for _, raw := range []string{
        `[{"type":"reasoning.text","text":"inspect state"},{"type":"reasoning.encrypted","data":"opaque"}]`,
        `[ { "type" : "reasoning.encrypted", "data" : "opaque", "index" : 0 } ]`,
        `[{"z":1e+02,"type":"reasoning.encrypted","data":"\\u0061","a":-0}]`,
        `[{"type":"reasoning.text","text":"hé🙂","text":"last"}]`,
        "[{\"type\":\"reasoning.text\",\"text\":\"<>&\u2028\u2029\"}]",
    } {
        body := `{"id":"gen-1","choices":[{"message":{"role":"assistant","reasoning":"inspect state","reasoning_details":` + raw + `,"content":"answer"},"finish_reason":"stop"}]}`
        var chat chatCompletionResponse
        if err := json.Unmarshal([]byte(body), &chat); err != nil { t.Fatal(err) }
        response, err := toAnthropicResponseFromChat(chat)
        if err != nil { t.Fatal(err) }
        bufferedDetails = append(bufferedDetails, map[string]any{"body":body, "details":raw, "signature":response.Content[0].Signature})
        chunks := []string{
            `{"choices":[{"index":0,"delta":{"reasoning":"inspect ","reasoning_details":` + raw + `}}]}`,
            `{"choices":[{"index":0,"delta":{"reasoning":"state","reasoning_details":` + raw + `}}]}`,
            `{"choices":[{"index":0,"delta":{"content":"answer"},"finish_reason":"stop"}]}`,
            `[DONE]`,
        }
        sse := ""
        for _, chunk := range chunks { sse += "data: " + chunk + "\n\n" }
        streamed := drainChatStream(t, sse)
        streamedDetails = append(streamedDetails, map[string]any{"chunks":chunks, "signature":streamed.Content[0].Signature, "details":string(decodeOpenRouterReasoningDetails(streamed.Content[0].Signature))})
    }
    golden["buffered_chat_details"] = bufferedDetails
    golden["streamed_chat_details"] = streamedDetails
    var reasoningGates []map[string]any
    for _, kind := range []string{"enabled", "disabled", "adaptive", "", " ENABLED "} {
        for _, budget := range []int{-1, 0, 2048} {
            for _, effort := range []string{"", " high "} {
                req := anthropic.CreateMessageRequest{Model: "gpt-5.6", MaxTokens: 8192,
                    Thinking: &anthropic.ThinkingConfig{Type: kind, BudgetTokens: budget}, ReasoningEffort: effort}
                chat, err := toChatRequest(req)
                if err != nil { t.Fatal(err) }
                response, err := toResponseParams(req)
                if err != nil { t.Fatal(err) }
                projections := map[string]any{}
                for name, value := range map[string]any{"chat": chat, "responses": response} {
                    encoded, err := json.Marshal(value)
                    if err != nil { t.Fatal(err) }
                    var body map[string]any
                    if err := json.Unmarshal(encoded, &body); err != nil { t.Fatal(err) }
                    projections[name] = body["reasoning"]
                }
                reasoningGates = append(reasoningGates, map[string]any{"type": kind, "budget": budget,
                    "effort": effort, "chat": projections["chat"], "responses": projections["responses"]})
            }
        }
    }
    golden["reasoning_gates"] = reasoningGates
    type toolOperation struct {
        Kind string `json:"kind"`
        ID string `json:"id"`
        Text string `json:"text"`
        Image bool `json:"image"`
    }
    toolCases := []struct { name string; operations []toolOperation }{
        {"orphan", []toolOperation{{"result", "call_missing", "/repo", false}, {"text", "", "continue", false}}},
        {"matched", []toolOperation{{"call", "call_1", "", false}, {"result", "call_1", "/repo", false}}},
        {"result-before-call", []toolOperation{{"result", "call_1", "/repo", false}, {"call", "call_1", "", false}}},
        {"trimmed-identifiers", []toolOperation{{"call", " call_1 ", "", false}, {"result", "\tcall_1\n", "/repo", false}}},
        {"generated-identifier", []toolOperation{{"call", " ", "", false}, {"result", "call_0", "/repo", false}}},
        {"blank-result-identifier", []toolOperation{{"call", "", "", false}, {"result", " ", "/repo", false}}},
        {"repeated-result", []toolOperation{{"call", "call_1", "", false}, {"result", "call_1", "first", false}, {"result", "call_1", "second", false}}},
        {"orphan-image", []toolOperation{{"result", "orphan", "inspect", true}, {"text", "", "continue", false}}},
        {"matched-image", []toolOperation{{"call", "image-call", "", false}, {"result", "image-call", "inspect", true}}},
        {"blank-output", []toolOperation{{"call", "call_1", "", false}, {"result", "call_1", " \t\n", false}}},
    }
    var toolPairs []map[string]any
    for _, tc := range toolCases {
        var messages []anthropic.Message
        for _, op := range tc.operations {
            message := anthropic.Message{Role: anthropic.RoleUser}
            switch op.Kind {
            case "call":
                message.Role = anthropic.RoleAssistant
                message.Content = []anthropic.ContentBlock{anthropic.NewToolUseBlock(op.ID, "Bash", []byte(`{"command":"pwd"}`))}
            case "result":
                message.Content = []anthropic.ContentBlock{anthropic.NewToolResultBlock(op.ID, op.Text, false)}
                if op.Image { message.Content = append(message.Content, anthropic.NewImageBlock("image/png", "cG5n")) }
            case "text":
                message.Content = []anthropic.ContentBlock{anthropic.NewTextBlock(op.Text)}
            }
            messages = append(messages, message)
        }
        items, err := toResponseInputItems(messages)
        if err != nil { t.Fatal(err) }
        toolPairs = append(toolPairs, map[string]any{"name": tc.name, "operations": tc.operations, "items": items})
    }
    golden["responses_tool_pairs"] = toolPairs
    var prices []map[string]any
    var models []string
    for model := range modelPricing { models = append(models, model) }
    models = append(models, "unknown-fixture")
    sort.Strings(models)
    for _, model := range models {
        for _, usage := range []anthropic.Usage{
            {InputTokens:1000,OutputTokens:100,CacheReadInputTokens:200,CacheCreationInputTokens:50},
            {InputTokens:300000,OutputTokens:100,CacheReadInputTokens:200000,CacheCreationInputTokens:50000},
            {InputTokens:10,OutputTokens:1,CacheReadInputTokens:20,CacheCreationInputTokens:20},
        } {
            cost, known := EstimateCost(model, usage)
            prices = append(prices, map[string]any{"model":model,"usage":usage,"cost":cost,"known":known})
        }
    }
    golden["openai_costs"] = prices
    var anthropicPrices []map[string]any
    for _, model := range []string{"claude-fable-5-1", "claude-fable-5", "claude-mythos-5-1", "claude-mythos-5", "claude-sonnet-5", "claude-sonnet-4-6", "claude-opus-4-6", "claude-opus-4-1", "claude-opus-4", "claude-haiku-4-5", "claude-haiku-3-5", "unknown-fixture"} {
        usage := anthropic.Usage{InputTokens:1000,OutputTokens:100,CacheReadInputTokens:200,CacheCreationInputTokens:50}
        anthropicPrices = append(anthropicPrices, map[string]any{"model":model,"usage":usage,"cost":anthropic.CalculateCost(model, usage)})
    }
    golden["anthropic_costs"] = anthropicPrices
    var retries []map[string]any
    for _, pair := range [][2]string{{"1",""},{"1500","9"},{"999999","9"},{"bad","12"},{"","-1"},{"","301"}} {
        h := http.Header{}
        h.Set("Retry-After-Ms", pair[0]); h.Set("Retry-After", pair[1])
        retries = append(retries, map[string]any{"milliseconds":pair[0],"seconds":pair[1],"capped_seconds":capRetryAfterSeconds(parseRetryAfterSeconds(h))})
    }
    golden["retry_headers"] = retries
    chatSSE := "data: " + `{"id":"chat-fixture","choices":[{"index":0,"delta":{"reasoning_content":"why ","content":"hé","tool_calls":[{"index":0,"id":"call-fixture","function":{"name":"lookup","arguments":"{\"q\":"}}]}}]}` + "\n\n" +
        "data: " + `{"id":"chat-fixture","choices":[{"index":0,"delta":{"content":"llo","tool_calls":[{"index":0,"function":{"arguments":"1}"}}]},"finish_reason":"tool_calls"}],"usage":{"prompt_tokens":100,"completion_tokens":20,"prompt_tokens_details":{"cached_tokens":70,"cache_write_tokens":11}}}` + "\n\ndata: [DONE]\n\n"
    reader := newChatStreamReader(&http.Response{Body:io.NopCloser(strings.NewReader(chatSSE))})
    assembler := anthropic.NewStreamAssembler()
    for {
        event, err := reader.Next()
        if err == io.EOF { break }
        if err != nil { t.Fatal(err) }
        assembler.Add(event)
    }
    if err := reader.Close(); err != nil { t.Fatal(err) }
    golden["chat_sse"] = chatSSE
    golden["chat_sse_response"] = assembler.Response()
    responseEvents := []json.RawMessage{
        json.RawMessage(`{"type":"response.created","response":{"id":"responses-fixture","model":"gpt-5.6"}}`),
        json.RawMessage(`{"type":"response.output_item.added","output_index":0,"item":{"type":"reasoning","id":"reasoning-fixture","summary":[]}}`),
        json.RawMessage(`{"type":"response.reasoning_summary_text.delta","output_index":0,"delta":"why"}`),
        json.RawMessage(`{"type":"response.output_item.done","output_index":0,"item":{"type":"reasoning","id":"reasoning-fixture","encrypted_content":"opaque"}}`),
        json.RawMessage(`{"type":"response.output_item.added","output_index":1,"item":{"type":"message","role":"assistant","content":[]}}`),
        json.RawMessage(`{"type":"response.content_part.added","output_index":1,"part":{"type":"output_text","text":""}}`),
        json.RawMessage(`{"type":"response.output_text.delta","output_index":1,"delta":"hé"}`),
        json.RawMessage(`{"type":"response.output_text.delta","output_index":1,"delta":"llo"}`),
        json.RawMessage(`{"type":"response.output_text.done","output_index":1,"text":"héllo"}`),
        json.RawMessage(`{"type":"response.output_item.added","output_index":2,"item":{"type":"function_call","id":"item-fixture","call_id":"call-fixture","name":"lookup","arguments":""}}`),
        json.RawMessage(`{"type":"response.function_call_arguments.done","output_index":2,"item_id":"item-fixture","arguments":"{\"q\":1}"}`),
        json.RawMessage(`{"type":"response.completed","response":{"status":"completed","end_turn":false,"output":[],"usage":{"input_tokens":100,"output_tokens":20,"input_tokens_details":{"cached_tokens":70,"cache_write_tokens":11}}}}`),
    }
    responseReader := &ResponsesStreamReader{}
    responseAssembler := anthropic.NewStreamAssembler()
    var responseSSE strings.Builder
    for _, raw := range responseEvents {
        var event responses.ResponseStreamEventUnion
        if err := event.UnmarshalJSON(raw); err != nil { t.Fatal(err) }
        responseReader.translateEvent(event)
        for _, translated := range responseReader.buf { responseAssembler.Add(translated) }
        responseReader.buf = nil
        responseSSE.WriteString("data: "); responseSSE.Write(raw); responseSSE.WriteString("\n\n")
    }
    golden["responses_sparse_sse"] = responseSSE.String()
    golden["responses_sparse_sse_response"] = responseAssembler.Response()
    data, err := json.MarshalIndent(golden, "", "  ")
    if err != nil { t.Fatal(err) }
    dir := os.Getenv("ADK_PROVIDER_FIXTURE_DIR")
    if dir == "" { t.Fatal("explicit fixture directory required") }
    if err := os.WriteFile(filepath.Join(dir, "continuation.json"), append(data, '\n'), 0644); err != nil { t.Fatal(err) }
}
