package anthropic

import (
	"encoding/json"
	"os"
	"path/filepath"
	"strings"
	"testing"
)

func TestADKAnthropicSchemaGolden(t *testing.T) {
	var rows []map[string]any
	inputs := []string{` { "x" : -0, "x":1e+02 } `, `{"x":"\ud800"}`, `{"x":1e999}`, "{\"x\":\"<>&\\u0061\u2028\u2029\"}", `null`, `[1,2]`, `"schema"`, `true`, `{"x":9007199254740993}`, `{"x":1e999,"x":1}`}
	for _, depth := range []int{127, 128, 9997, 9998, 9999, 10000, 10001} {
		inputs = append(inputs, strings.Repeat("[", depth)+"0"+strings.Repeat("]", depth))
	}
	for _, raw := range inputs {
		params, _ := toSDKParams(&CreateMessageRequest{Model: "fixture", MaxTokens: 100, OutputSchema: &OutputSchema{Schema: json.RawMessage(raw)}})
		_, err := params.MarshalJSON()
		row := map[string]any{"raw": raw, "error": err != nil}
		if err == nil {
			body, err := params.OutputFormat.MarshalJSON()
			if err != nil {
				t.Fatal(err)
			}
			var format map[string]json.RawMessage
			if err := json.Unmarshal(body, &format); err != nil {
				t.Fatal(err)
			}
			row["schema_json"] = string(format["schema"])
		}
		rows = append(rows, row)
	}
	encoded, err := json.MarshalIndent(rows, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(os.Getenv("ADK_PROVIDER_FIXTURE_DIR"), "anthropic-schemas.json"), append(encoded, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
}
